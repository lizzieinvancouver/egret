		\frame{
		\frametitle{How the model works}
			\begin{align*}
		p(\mu_i | \theta) &= \begin{cases} 
   1 -  logit^{-1}(\mu_i - k_1) & \text{if } \mu_i = 0 \\
    [logit^{-1}(\mu_i - k_1) - logit^{-1}(\mu_i - k_2)] \\
    \times \text{Beta\_proportion}(logit^{-1}(\mu_i), k_1) & \text{if } \mu_i \in (0, 1) \\
    logit^{-1}(\mu_i - k_2) & \text{if } \mu_i = 1
\end{cases} \\
		\mu_i &= \alpha_{[sp]} +  \beta_{f[sp]}  \cdot \text{forcing} +  \beta_{cs[sp]} \cdot \text{chilling} \\
			\end{align*}
	}
